cmake_minimum_required(VERSION 3.16)
project(BcoGenerator VERSION 0.1.0 LANGUAGES CXX)

# Portable core has no dependency on Qt, OS networking, or SiriusScope.
add_library(bco_generator STATIC src/generator.cpp src/protocol.cpp)
add_library(BcoGenerator::Core ALIAS bco_generator)
set_target_properties(bco_generator PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_compile_features(bco_generator PUBLIC cxx_std_20)
target_include_directories(bco_generator PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

set(_bco_standalone OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(_bco_standalone ON)
endif()
option(BCO_GENERATOR_BUILD_TESTS "Build portable generator tests" ${_bco_standalone})
if(BCO_GENERATOR_BUILD_TESTS)
    enable_testing()
    add_executable(tst_bco_protocol tests/tst_protocol.cpp)
    set_target_properties(tst_bco_protocol PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(tst_bco_protocol PRIVATE BcoGenerator::Core)
    add_test(NAME tst_bco_protocol COMMAND tst_bco_protocol)
    add_executable(tst_bco_generator tests/tst_generator.cpp)
    set_target_properties(tst_bco_generator PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(tst_bco_generator PRIVATE BcoGenerator::Core)
    add_test(NAME tst_bco_generator COMMAND tst_bco_generator)
endif()

option(BCO_GENERATOR_BUILD_HOST "Build host UDP transport and executable" ON)
if(BCO_GENERATOR_BUILD_HOST)
    find_package(Threads REQUIRED)
    add_library(bco_generator_host STATIC src/host/udp_socket.cpp src/host/udp_generator.cpp)
    add_library(BcoGenerator::Host ALIAS bco_generator_host)
    set_target_properties(bco_generator_host PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(bco_generator_host PUBLIC BcoGenerator::Core Threads::Threads)
    if(WIN32)
        target_link_libraries(bco_generator_host PUBLIC ws2_32)
    endif()
    add_executable(bco-generator apps/bco_generator_main.cpp)
    set_target_properties(bco-generator PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    target_link_libraries(bco-generator PRIVATE BcoGenerator::Host)
endif()
